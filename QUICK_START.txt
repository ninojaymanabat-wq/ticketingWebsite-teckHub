                      TRANSACTION TABLE SCHEMA
================================================================================

Columns (16 total):
  id                    - Primary Key
  user_id              - Foreign Key to User
  booking_type         - 'movie' or 'bus'
  booking_id           - Reference to booking
  amount               - Payment amount (PHP)
  currency             - Currency code
  payment_method       - 'gcash' or 'paymaya'

  paymongo_payment_id  - PayMongo payment ID
  transaction_reference - Unique transaction code
  error_message        - Error details if failed
  error_code           - Error code from PayMongo
  created_at           - Creation timestamp
  updated_at           - Last update timestamp
  completed_at         - Completion timestamp

================================================================================
                       TROUBLESHOOTING
================================================================================

Issue: "Database is locked"
Fix: Delete ticketing.db and run: python init_db.py

Issue: "PayMongo API error"
Fix: 1) Check .env for correct API keys
     2) Verify internet connection
     3) Check error_message in Transaction table

Issue: "Payment not updating"
Fix: 1) Check transaction status in database
     2) Verify PayMongo webhook settings
     3) Check redirect URLs are correct

Issue: "Import Error: No module named 'requests'"
Fix: Run: pip install -r requirements.txt

Issue: "Secret Key error"
Fix: Change SECRET_KEY in .env file

================================================================================
                        QUICK SQL QUERIES
================================================================================

# Get all transactions
SELECT * FROM transaction ORDER BY created_at DESC;

# Get completed transactions
SELECT * FROM transaction WHERE payment_status='completed';

# Get failed transactions with errors
SELECT * FROM transaction 
  WHERE payment_status='failed' AND error_message IS NOT NULL;

# Calculate total revenue
SELECT SUM(amount) as total_revenue FROM transaction 
  WHERE payment_status='completed';

# Get transactions by payment method
SELECT payment_method, COUNT(*) as count, SUM(amount) as total
  FROM transaction WHERE payment_status='completed'
  GROUP BY payment_method;

# Get today's transactions
SELECT * FROM transaction 
  WHERE DATE(created_at) = CURDATE()
  ORDER BY created_at DESC;
